1"""Methods calibration numbers: the per-cell envelope test's coverage check on its own nulls. 

2 

3The ``calibration_summary`` step of ``run_manuscript_workflow`` reads each structure's 

4calibration cache (``per_cell_envelope_results_calibration.dat``, written by 

5``run_per_cell_envelope_test`` with ``calibration_holdout_index`` set) and writes 

6``Methods/Methods_numbers.md`` and ``Methods/calibration_summary.csv``. 

7 

8In that run one replicate of every cell and null model is the pseudo-observed curve and the 

9envelope comes from the other ``R`` replicates. Each p-value is then a multiple of ``1/(R+1)``, 

10so a calibrated test rejects at p < alpha at the rate ``expected_raw`` (4/99 for R = 98, 

11alpha = 0.05) and, after Benjamini-Hochberg correction across cells, essentially never. 

12 

13``R`` is not stored in the cache. It is measured from the lattice the p-values lie on and 

14checked against the config's ``expected.n_replicates``; a wrong ``R`` would shift the 

15at-floor rate silently. 

16"""

59def measure_envelope_replicates(per_cell_results: dict[str, Any]) -> int: 

60 """Measure the envelope's replicate count ``R`` from the lattice of the p-values. 

61 

62 Every Monte Carlo p-value is ``(1 + #{T_sim >= T_obs}) / (R + 1)``, so all of them are 

63 multiples of ``1/(R+1)``. The result is the smallest such ``R`` over every per-measure and 

64 joint p-value, which equals the true ``R`` unless every count shares a factor. 

65 

66 Parameters 

67 ---------- 

68 per_cell_results 

69 Return value of ``per_cell_envelope_test`` (as pickled by the pipeline). 

70 

71 Returns 

72 ------- 

73 : 

74 The measured ``R``. 

75 

76 Raises 

77 ------ 

78 ValueError 

79 If a p-value is missing or outside (0, 1], or the p-values do not lie on one lattice 

80 (cells with different replicate counts). 

81 """ 

82 summary = per_cell_results["summary"] 

83 p = np.concatenate( 

84 [ 

85 np.asarray(summary["per_distance_measure_pvals"], dtype=float).ravel(), 

86 np.asarray(summary["joint_pvals"], dtype=float).ravel(), 

87 ] 

88 ) 

89 if p.size == 0 or not np.all(np.isfinite(p)) or p.min() <= 0 or p.max() > 1: 

90 raise ValueError("calibration p-values must be finite and in (0, 1]") 

91 for n in range(2, MAX_ENVELOPE_REPLICATES + 2): 

92 scaled = p * n 

93 if np.all(np.abs(scaled - np.rint(scaled)) < LATTICE_ATOL): 

94 return n - 1 

95 raise ValueError( 

96 f"calibration p-values are not multiples of one 1/(R+1) with R <= " 

97 f"{MAX_ENVELOPE_REPLICATES}; the cells' replicate counts differ" 

98 )

101def check_calibration_result( 

102 config: ManuscriptConfig, structure: str, result: dict[str, Any] 

103) -> int: 

104 """Check a calibration result against the config and return its measured ``R``. 

105 

106 Parameters 

107 ---------- 

108 config 

109 The manuscript config (``expected`` holds alpha, replicates and cell counts). 

110 structure 

111 Structure (packing id) the result belongs to. 

112 result 

113 Return value of ``per_cell_envelope_test`` run with ``calibration_holdout_index``. 

114 

115 Returns 

116 ------- 

117 : 

118 The measured replicate count ``R`` of each envelope. 

119 

120 Raises 

121 ------ 

122 ValueError 

123 If the result is not a calibration run, or its alpha, cell count, null modes, distance 

124 measures or measured ``R`` differ from the config's expected values. 

125 """ 

126 what = f"{structure} calibration result" 

127 expected = config.expected 

128 if result.get("calibration_holdout_index") is None: 

129 raise ValueError(f"{what} is not a hold-one-out calibration run") 

130 alpha = expected["envelope"]["alpha"] 

131 if result.get("alpha") != alpha: 

132 raise ValueError(f"{what}: alpha {result.get('alpha')!r} != expected {alpha!r}") 

133 n_cells = len(result["cell_ids"]) 

134 if n_cells != expected["n_cells"][structure]: 

135 raise ValueError(f"{what}: {n_cells} cells != expected {expected['n_cells'][structure]}") 

136 if sorted(result["distance_measures"]) != sorted(config.distance_measures): 

137 raise ValueError( 

138 f"{what}: distance measures {result['distance_measures']} != {config.distance_measures}" 

139 ) 

140 if not result["null_modes"]: 

141 raise ValueError(f"{what} has no null models") 

142 measured = measure_envelope_replicates(result) 

143 # one replicate is held out of its own envelope 

144 want = expected["n_replicates"] - 1 

145 if measured != want: 

146 raise ValueError( 

147 f"{what}: measured R = {measured} (p-values are multiples of 1/{measured + 1}), " 

148 f"expected {want} = {expected['n_replicates']} replicates - 1 held out" 

149 ) 

150 return measured

153def calibration_table( 

154 config: ManuscriptConfig, structure: str, result: dict[str, Any] 

155) -> pd.DataFrame: 

156 """Tidy calibration rows for one structure (one row per null model and scope). 

157 

158 Parameters 

159 ---------- 

160 config 

161 The manuscript config. 

162 structure 

163 Structure (packing id) the result belongs to. 

164 result 

165 Return value of ``per_cell_envelope_test`` run with ``calibration_holdout_index``. 

166 

167 Returns 

168 ------- 

169 : 

170 ``stats.summarize_per_cell_calibration`` rows plus ``structure``, ``n_cells``, 

171 ``n_envelope_replicates``, ``holdout_index`` and the integer cell counts 

172 ``n_raw_rejected`` and ``n_bh_rejected``; columns are ``TABLE_COLUMNS``. 

173 """ 

174 r = check_calibration_result(config, structure, result) 

175 table = summarize_per_cell_calibration(result, r, alpha=result["alpha"]) 

176 n_cells = len(result["cell_ids"]) 

177 table["structure"] = structure 

178 table["n_cells"] = n_cells 

179 table["n_envelope_replicates"] = r 

180 table["holdout_index"] = int(result["calibration_holdout_index"]) 

181 for column, count in (("raw_rejection", "n_raw_rejected"), ("bh_rejection", "n_bh_rejected")): 

182 scaled = table[column].to_numpy(dtype=float) * n_cells 

183 if not np.allclose(scaled, np.rint(scaled), atol=1e-6): 

184 raise ValueError(f"{structure}: {column} is not a fraction of {n_cells} cells") 

185 table[count] = np.rint(scaled).astype(int) 

186 return table[TABLE_COLUMNS].reset_index(drop=True)

189def calibration_tables(config: ManuscriptConfig, sources: PaperSources) -> pd.DataFrame: 

190 """Calibration rows for every structure of the config, from the calibration caches. 

191 

192 Parameters 

193 ---------- 

194 config 

195 The manuscript config. 

196 sources 

197 Cache resolver; ``per_cell_envelope(structure, "_calibration")`` is read. 

198 

199 Returns 

200 ------- 

201 : 

202 The structures' :func:`calibration_table` rows, in config order. 

203 """ 

204 tables = [] 

205 for structure in config.structures: 

206 result = sources.per_cell_envelope(structure, CALIBRATION_SUFFIX) 

207 tables.append(calibration_table(config, structure, result)) 

208 return pd.concat(tables, ignore_index=True)

222def range_table(config: ManuscriptConfig, table: pd.DataFrame) -> pd.DataFrame: 

223 """Min, max and mean of the raw rejection rate over each organelle and scope set. 

224 

225 Parameters 

226 ---------- 

227 config 

228 The manuscript config (row labels for the argmin and argmax). 

229 table 

230 Output of :func:`calibration_tables`. 

231 

232 Returns 

233 ------- 

234 : 

235 One row per organelle (each structure, then ``"both"``) and scope set (``SCOPE_SETS``): 

236 ``organelles``, ``scopes``, ``n_rows``, ``min``, ``min_at``, ``max``, ``max_at``, ``mean``. 

237 """ 

238 groups = [*dict.fromkeys(table["structure"]), "both"] 

239 rows = [] 

240 for organelles in groups: 

241 in_group = table if organelles == "both" else table[table["structure"] == organelles] 

242 for scope_set in SCOPE_SETS: 

243 sub = in_group[_in_set(in_group["scope"], scope_set)] 

244 raw = sub["raw_rejection"] 

245 rows.append( 

246 { 

247 "organelles": organelles, 

248 "scopes": scope_set, 

249 "n_rows": len(sub), 

250 "min": raw.min(), 

251 "min_at": _row_label(config, sub.loc[raw.idxmin()]), 

252 "max": raw.max(), 

253 "max_at": _row_label(config, sub.loc[raw.idxmax()]), 

254 "mean": raw.mean(), 

255 } 

256 ) 

257 return pd.DataFrame(rows)

270 "",

274def methods_markdown( 

275 config: ManuscriptConfig, table: pd.DataFrame, inputs: list[Path] | None = None 

276) -> str: 

277 """Render the Methods calibration numbers sheet. 

278 

279 Parameters 

280 ---------- 

281 config 

282 The manuscript config. 

283 table 

284 Output of :func:`calibration_tables`. 

285 inputs 

286 Files the numbers were read from (listed with their modification times). 

287 

288 Returns 

289 ------- 

290 : 

291 The markdown text of ``Methods/Methods_numbers.md``. 

292 """ 

293 alpha = config.expected["envelope"]["alpha"] 

294 r = int(table["n_envelope_replicates"].iloc[0]) 

295 expected_raw = float(table["expected_raw"].iloc[0]) 

296 n_below = round(expected_raw * (r + 1)) 

297 ranges = range_table(config, table) 

298 both = ranges[ranges["organelles"] == "both"].set_index("scopes") 

299 n_rows = len(table) 

300 n_bh_rows = int((table["n_bh_rejected"] == 0).sum()) 

301 n_bh_cells = int(table["n_bh_rejected"].sum()) 

302 n_tests = int(table["n_cells"].sum()) 

303 lo, hi = both.loc["all scopes", "min"], both.loc["all scopes", "max"] 

304 per_measure = both.loc["per measure"] 

305 same = (_g(per_measure["min"]), _g(per_measure["max"])) == (_g(lo), _g(hi)) 

306 structures = list(dict.fromkeys(table["structure"])) 

307 cells = " / ".join( 

308 f"{int(table[table['structure'] == s]['n_cells'].iloc[0])}" for s in structures 

309 ) 

310 measures = [s for s in dict.fromkeys(table["scope"]) if s != "joint"] 

311 holdout = int(table["holdout_index"].iloc[0]) 

312 holdout_note = ( 

313 f"Replicate {holdout} of each cell and null model is the pseudo-observed curve and is " 

314 f"tested against the other R = {r} replicates " 

315 f"({config.expected['n_replicates']} per cell and rule, one held out)." 

316 ) 

317 after = ( 

318 "no cells were rejected in this calibration after correction" 

319 if n_bh_cells == 0 

320 else f"{n_bh_cells} of {n_tests} cell-level tests were rejected after correction" 

321 ) 

322 envelope = config.expected["envelope"] 

323 lines = [ 

324 f"# Methods (envelope calibration) — numbers for the text ({config.condition})", 

325 "", 

326 "Written by `run_manuscript_workflow` (step `calibration_summary`) from the calibration " 

327 "caches listed below; nothing is typed by hand. Rates are fractions of cells.", 

328 "", 

329 f"**Provenance.** Engine pin `{engine_commit()[:8]}`; analysis code `{code_version()}`; " 

330 f"per-cell envelope test, statistic {envelope['statistic']}, " 

331 f"{envelope['envelope_type']} envelope, " 

332 f"alpha = {alpha:g}, run as a hold-one-out calibration. {holdout_note} " 

333 f"{' / '.join(structures)}: n = {cells} cells (asserted); R is measured from the " 

334 f"p-value lattice (every p is a multiple of 1/{r + 1}) and asserted equal to the " 

335 f"configured replicates minus one.", 

336 "", 

337 ] 

338 if inputs: 

339 rows = [[f"`{_rel(config, p)}`", _modified(p)] for p in inputs] 

340 lines += _md_table(["Source", "Modified (UTC)"], rows) 

341 lines += [ 

342 "## Envelope calibration (Methods)", 

343 "", 

344 f"Uncorrected rejection rates at p < {alpha:g} were {_g(lo)}\N{EN DASH}{_g(hi)} across " 

345 f"modeled rules, distance measures, and organelles (expected: {_g(expected_raw)}), and " 

346 f"{after}.", 

347 "", 

348 f"Expected under a calibrated test: raw rate {expected_raw:.4f} " 

349 f"({n_below}/{r + 1}, p in {{1/{r + 1}, ..., {n_below}/{r + 1}}}), BH rate 0, " 

350 f"at-floor rate {1 / (r + 1):.4f} (1/{r + 1}).", 

351 "", 

352 f"**Scopes.** The range above spans all {n_rows} rows: {len(structures)} organelles " 

353 f"\N{MULTIPLICATION SIGN} {table['null_mode'].nunique()} rules \N{MULTIPLICATION SIGN} " 

354 f"{len(measures) + 1} scopes ({', '.join(measures)} and joint). " 

355 + ( 

356 f"The {int(per_measure['n_rows'])} per-measure rows alone give the same range." 

357 if same 

358 else f"The {int(per_measure['n_rows'])} per-measure rows alone give " 

359 f"{_g(per_measure['min'])}\N{EN DASH}{_g(per_measure['max'])}." 

360 ), 

361 "", 

362 "### Raw rejection rate by scope set", 

363 "", 

364 ] 

365 rows = [] 

366 for rec in ranges.itertuples(): 

367 rows.append( 

368 [ 

369 str(rec.organelles), 

370 str(rec.scopes), 

371 str(rec.n_rows), 

372 f"{rec.min:.4f}", 

373 rec.min_at, 

374 f"{rec.max:.4f}", 

375 rec.max_at, 

376 f"{rec.mean:.4f}", 

377 ] 

378 ) 

379 lines += _md_table( 

380 ["Organelles", "Scopes", "Rows", "Min", "Min at", "Max", "Max at", "Mean"], rows 

381 ) 

382 lines += [ 

383 "### Rejection after BH correction", 

384 "", 

385 f"BH q < {alpha:g} rejected {n_bh_cells} of {n_tests} cell-level tests; the rate is 0 in " 

386 f"{n_bh_rows} of {n_rows} rows. The smallest p-value anywhere is " 

387 f"{table['min_p'].min():.4f} (floor {1 / (r + 1):.4f}). Because p >= {1 / (r + 1):.4f}, " 

388 f"any BH rejection at alpha {alpha:g} would reject at least " 

389 f"{1 / (r + 1) / alpha:.0%} of a row's cells.", 

390 "", 

391 "### All rows", 

392 "", 

393 ] 

394 for structure in structures: 

395 sub = table[table["structure"] == structure] 

396 lines += [f"**{structure} (n = {int(sub['n_cells'].iloc[0])}):**", ""] 

397 rows = [ 

398 [ 

399 config.row_labels.get(rec.null_mode, rec.null_mode), 

400 str(rec.scope), 

401 f"{rec.raw_rejection:.4f} ({rec.n_raw_rejected})", 

402 f"{rec.bh_rejection:.4f} ({rec.n_bh_rejected})", 

403 f"{rec.at_floor:.4f}", 

404 f"{rec.min_p:.4f}", 

405 ] 

406 for rec in sub.itertuples() 

407 ] 

408 lines += _md_table( 

409 [ 

410 "Null model", 

411 "Scope", 

412 f"Raw p < {alpha:g} (cells)", 

413 f"BH q < {alpha:g} (cells)", 

414 "At floor", 

415 "Min p", 

416 ], 

417 rows, 

418 ) 

419 return "\n".join(lines).rstrip() + "\n"

434def write_calibration_summary(config: ManuscriptConfig, sources: PaperSources) -> list[Path]: 

435 """Write ``Methods/Methods_numbers.md`` and ``Methods/calibration_summary.csv``. 

436 

437 This is the ``calibration_summary`` step of ``run_manuscript_workflow``. 

438 

439 Parameters 

440 ---------- 

441 config 

442 The manuscript config. 

443 sources 

444 Cache resolver; the calibration caches are read through it. 

445 

446 Returns 

447 ------- 

448 : 

449 The two files written, under ``config.results_dir``. 

450 """ 

451 with sources.record_inputs() as inputs: 

452 table = calibration_tables(config, sources) 

453 out_dir = check_under(config.results_dir / OUTPUT_DIR, config.results_dir, "Methods folder") 

454 out_dir.mkdir(parents=True, exist_ok=True) 

455 sheet = out_dir / SHEET_NAME 

456 csv = out_dir / CSV_NAME 

457 sheet.write_text(methods_markdown(config, table, inputs), encoding="utf-8") 

458 table.to_csv(csv, index=False, lineterminator="\n") 

459 logger.info("Wrote %s and %s (%d rows)", sheet, csv, len(table)) 

460 return [sheet, csv]
