10def validate_structure_stats_units(df: pd.DataFrame) -> None: 

11 """

13 

14 Parameters 

15 ---------- 

16 df 

17 The full structure stats DataFrame. 

18 

19 Raises

21 ValueError 

22 If the table has the legacy layout (``sphericity``, no ``mem_sphericity``) or the median 

23 radius of SLC25A17 or RAB5A is at least 1.0, i.e. it is in pixels. 

24 """ 

25 reasons = [] 

26 if "mem_sphericity" not in df.columns: 

27 reasons.append("missing column 'mem_sphericity'") 

28 if "sphericity" in df.columns: 

29 reasons.append("has the legacy column 'sphericity'") 

30 for structure_id in CHECKED_STRUCTURES: 

31 radius = df.loc[df["structure_name"] == structure_id, "radius"] 

32 if len(radius) and radius.median() >= MAX_MEDIAN_RADIUS_UM: 

33 reasons.append(f"median radius of {structure_id} is {radius.median():.3f} (>= 1.0)") 

34 if reasons: 

35 raise ValueError( 

36 f"pixel-unit structure_stats; fetch the release tables ({FETCH_TABLES_COMMAND}). " 

37 f"Problems: {'; '.join(reasons)}"
